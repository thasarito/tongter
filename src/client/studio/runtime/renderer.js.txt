  }
  setLayout(state,selected,opts,warnings=[]){this.state=state;this.selected=selected;this.opts=opts;this.clearGroup(this.groups.furniture);this.clearGroup(this.groups.labels);this.clearGroup(this.groups.seatLabels);this.itemGroups.clear();

  }

  setView(mode){if(mode!==this.mode)this.stopWalking();this.mode=mode;this.lastFrame=0;this.setOptions(this.opts);this.resize();this.updateCamera();this.invalidate();}
  updateCamera(){
    if(this.mode==='inside'){const w=this.walk;this.camera.position.set(w.x,1.67,w.z);this.camera.lookAt(w.x+Math.sin(w.yaw)*Math.cos(w.pitch),1.67+Math.sin(w.pitch),w.z+Math.cos(w.yaw)*Math.cos(w.pitch));}
    else{const a=this.orbit,r=a.radius;this.camera.position.set(a.target.x+r*Math.sin(a.phi)*Math.sin(a.theta),a.target.y+r*Math.cos(a.phi),a.target.z+r*Math.sin(a.phi)*Math.cos(a.theta));this.camera.lookAt(a.target);}

    this.invalidate();
  }

  seatAt(event){this.rayAt(event);const hit=this.ray.intersectObjects(this.groups.furniture.children,true)[0];if(!hit)return null;const number=hit.object.userData.seatNumber||(hit.instanceId!==undefined?hit.object.userData.seatNumbers?.[hit.instanceId]:null);return number?{tableId:hit.object.userData.itemId,seatNumber:number}:null;}
