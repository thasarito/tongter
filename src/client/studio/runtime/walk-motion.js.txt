/* Continuous eye-level motion. Renderer independent; metres and seconds.
   Exact exponential velocity integration keeps the same pace at 30/60/144 Hz.
   Small collision substeps slide along boundaries without camera bobbing. */
(function(root){
'use strict';
class WalkMotion {
  constructor({x=-11.2,z=-1.65,yaw=Math.PI/2,pitch=.035,canMove=()=>true}={}) {
    Object.assign(this,{x,z,yaw,pitch,canMove,targetYaw:yaw,targetPitch:pitch,vx:0,vz:0,speed:1.6,fast:false});
    this.inputs=new Map();
  }
  setInput(key,right,forward){this.inputs.set(key,{right,forward});}
  removeInput(key){this.inputs.delete(key);}
  look(yaw,pitch){this.targetYaw+=yaw;this.targetPitch=Math.max(-1.05,Math.min(1.15,this.targetPitch+pitch));}
  stop(){this.inputs.clear();this.vx=this.vz=0;this.fast=false;this.targetYaw=this.yaw;this.targetPitch=this.pitch;}
  reset(){this.stop();this.x=-11.2;this.z=-1.65;this.yaw=this.targetYaw=Math.PI/2;this.pitch=this.targetPitch=.035;}
  get active(){return this.inputs.size>0||Math.abs(this.vx)+Math.abs(this.vz)>.001||Math.abs(this.targetYaw-this.yaw)+Math.abs(this.targetPitch-this.pitch)>.00001;}
  update(delta){
    if(!Number.isFinite(delta)||delta<=0)return false;
    const total=Math.min(delta,.05),steps=Math.ceil(total/(1/120)),dt=total/steps;
    const old=[this.x,this.z,this.yaw,this.pitch];
    let right=0,forward=0;for(const v of this.inputs.values()){right+=v.right;forward+=v.forward;}
    const length=Math.hypot(right,forward);if(length>1){right/=length;forward/=length;}
    for(let i=0;i<steps;i++){
      const look=1-Math.exp(-22*dt);
      this.yaw+=(this.targetYaw-this.yaw)*look;this.pitch+=(this.targetPitch-this.pitch)*look;
      if(Math.abs(this.targetYaw-this.yaw)<.00001)this.yaw=this.targetYaw;
      if(Math.abs(this.targetPitch-this.pitch)<.00001)this.pitch=this.targetPitch;
      const speed=this.speed*(this.fast?1.65:1),sin=Math.sin(this.yaw),cos=Math.cos(this.yaw);
      const tx=(sin*forward-cos*right)*speed,tz=(cos*forward+sin*right)*speed;
      const lambda=length?12:18,decay=Math.exp(-lambda*dt);
      const dx=tx*dt+(this.vx-tx)*(1-decay)/lambda,dz=tz*dt+(this.vz-tz)*(1-decay)/lambda;
      this.vx=tx+(this.vx-tx)*decay;this.vz=tz+(this.vz-tz)*decay;
      if(this.canMove(this.x+dx,this.z+dz)){this.x+=dx;this.z+=dz;}
      else {
        if(this.canMove(this.x+dx,this.z))this.x+=dx;else this.vx=0;
        if(this.canMove(this.x,this.z+dz))this.z+=dz;else this.vz=0;
      }
      if(!length&&Math.abs(this.vx)<.001)this.vx=0;
      if(!length&&Math.abs(this.vz)<.001)this.vz=0;
    }
    return old.some((v,i)=>v!==[this.x,this.z,this.yaw,this.pitch][i]);
  }
}
root.WalkMotion=WalkMotion;
if(typeof module!=='undefined'&&module.exports)module.exports=WalkMotion;
})(typeof window!=='undefined'?window:globalThis);
