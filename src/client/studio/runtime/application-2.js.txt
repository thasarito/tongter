function preserveRosterOnLayout(next){const n=M.clone(next);const ids=new Set(n.items.filter(o=>o.kind==='table').map(o=>o.id));n.guestList=M.clone(state.guestList);n.guestSource=M.clone(state.guestSource);for(const g of n.guestList){const table=n.items.find(t=>t.id===g.tableId&&t.kind==='table');if(!ids.has(g.tableId)||table&&g.seatNumber>table.seats){g.tableId='';g.seatNumber=null;}}const existingIds=new Set(n.guestList.map(g=>g.id));for(const g of next.guestList||[])if(!existingIds.has(g.id))n.guestList.push({...g,tableId:'',seatNumber:null});return G.normalizeState(n);}
