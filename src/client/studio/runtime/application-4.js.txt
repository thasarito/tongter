function planSeatAtPoint(x,y){
 const svg=$('plan-svg');if(!svg||!options.chairs||!options.tables)return null;
 const r=svg.getBoundingClientRect();if(x<r.left||x>r.right||y<r.top||y>r.bottom)return null;
 const p=planAt({clientX:x,clientY:y},svg);let best=null,distance=.30*.30;
 for(const t of tables())for(const slot of S.slots(state,t.id)){const d=(p.x-slot.worldX)**2+(p.y-slot.worldZ)**2;if(d<distance){distance=d;best={tableId:t.id,seatNumber:slot.number};}}
 return best;
}
function seatOccupant(tableId,number){return state.guestList.find(g=>g.tableId===tableId&&g.seatNumber===number)||null;}
function seatDescription(tableId,number){const g=seatOccupant(tableId,number);return `${tableLabel(tableId)} · Seat ${number}\n${g?g.name:'Empty · click to assign'}${g?'\n'+[g.host,g.group].filter(Boolean).join(' · '):''}`;}
function showSeatTip(tableId,number,x,y){const tip=$('seat-tooltip');if(!tableId||guestDrag?.active){tip.hidden=true;return;}tip.textContent=seatDescription(tableId,number);tip.hidden=false;tip.style.left=Math.max(8,Math.min(innerWidth-tip.offsetWidth-10,x+15))+'px';tip.style.top=Math.max(8,Math.min(innerHeight-tip.offsetHeight-10,y+18))+'px';}
function hideSeatTip(){$('seat-tooltip').hidden=true;}
function chooseSeat(number){const t=tables().find(t=>t.id===seatTableId);if(!t)return;seatNumber=Math.max(1,Math.min(t.seats,number));seatPickedId=null;setGuestError('seat-map-error','');renderSeatMap();}

function seatDiagramSVG(t,number=seatNumber){
 const slots=S.slots(state,t.id),dx=slots.map(s=>s.worldX-t.x),dz=slots.map(s=>s.worldZ-t.z);
 const names=options.guestNames?L.diagram(state,t.id,text=>measureName(text,11.5)):null;
 const scale=names?names.scale:Math.min(155/(Math.max(...dx.map(Math.abs))+.28),93/(Math.max(...dz.map(Math.abs))+.28));
 const cx=245,cy=names?names.centerY:142,height=names?names.height:270;
 const closest=slots.length>1?Math.min(...slots.flatMap((a,i)=>slots.slice(i+1).map(b=>Math.hypot(a.x-b.x,a.z-b.z))))*scale:100;
 const half=Math.min(17,closest*.34),hitRadius=Math.min(25,closest*.47),fontSize=Math.min(14,Math.max(8,half*.86));
 const tableTop=t.shape==='rect'?`<rect x="${-t.w/2*scale}" y="${-t.d/2*scale}" width="${t.w*scale}" height="${t.d*scale}" rx="8"/>`:`<ellipse rx="${t.w/2*scale}" ry="${t.d/2*scale}"/>`;
 let out=`<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 490 ${height}" role="group" aria-label="${escape(tableLabel(t.id))}, chairs and assigned guest names"><text x="14" y="18" font-size="9" fill="#8b957e" letter-spacing="1">↑ TOP OF FLOOR PLAN</text><g transform="translate(${cx} ${cy}) rotate(${t.rotation})" fill="#e3e9d7" stroke="#b2c0a1" stroke-width="1.5">${tableTop}</g><text x="${cx}" y="${cy-9}" text-anchor="middle" font-size="10" letter-spacing="2" fill="#738366">TABLE</text><text x="${cx}" y="${cy+21}" text-anchor="middle" font-family="Georgia,serif" font-size="26" fill="#3b5a42">${escape(t.label)}</text>`;
 if(names)out+=names.labels.map(l=>nameBadgeSVG(l,{detail:true,tableId:t.id,active:l.number===number})).join('');
 for(const s of slots){const x=cx+(s.worldX-t.x)*scale,y=cy+(s.worldZ-t.z)*scale,active=s.number===number;
 out+=`<g class="seat-detail-chair ${s.guest?'occupied':'empty'}" data-seat-select="${s.number}" data-guest-drop-table="${escape(t.id)}" data-guest-drop-seat="${s.number}" ${s.guest?`data-drag-guest="${escape(s.guest.id)}"`:''} role="button" tabindex="0" aria-label="Seat ${s.number}: ${escape(s.guest?.name||'Empty')}" aria-pressed="${active}" transform="translate(${x} ${y})"><title>${escape(seatDescription(t.id,s.number))}</title><circle r="${hitRadius}" fill="transparent"/><rect class="detail-chair-shape" x="${-half}" y="${-half}" width="${half*2}" height="${half*2}" rx="${half*.55}" fill="${s.guest?'#37654d':'#fffdf4'}" stroke="${active?'#b98837':s.guest?'#37654d':'#acb69a'}" stroke-width="${active?4:1.4}"/><text text-anchor="middle" y="${fontSize*.35}" font-size="${fontSize}" font-weight="600" fill="${s.guest?'#ffffff':'#637b53'}" pointer-events="none">${s.number}</text>${s.guest?`<circle cx="${half*.8}" cy="${-half*.88}" r="${Math.min(5,half*.3)}" fill="#b5c9a4" stroke="#fffdf4" stroke-width="1"/>`:''}</g>`;
 }return out+'</svg>';
}

function applySeatGuest(replace=false){
