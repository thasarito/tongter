function tableGuests(s,id){return (s.guestList||[]).filter(g=>g.tableId===id).sort((a,b)=>(a.seatNumber||999)-(b.seatNumber||999)||a.name.localeCompare(b.name));}

function csvParse(text){

 for(let i=0;i<text.length;i++){const c=text[i];if(quoted){if(c==='"'){if(text[i+1]==='"'){v+='"';i++;}else{quoted=false;closed=true;}}else v+=c;continue;}

 if(c==='"'){if(v||closed)throw new Error('Invalid CSV quote.');quoted=true;}else if(c===','){row.push(v);v='';closed=false;}else if(c==='\n'||c==='\r'){if(c==='\r'&&text[i+1]==='\n')i++;row.push(v);if(row.some(x=>x!==''))rows.push(row);row=[];v='';closed=false;}else v+=c;
 }if(quoted)throw new Error('Invalid CSV: unclosed quote.');if(v||row.length||closed){row.push(v);if(row.some(x=>x!==''))rows.push(row);}return rows;
}

function parseGuestFile(text,filename,s){
 if(typeof text!=='string'||text.length>5000000)throw new Error('Guest files must be smaller than 5 MB.');
 if(text.includes('\uFFFD'))throw new Error('Invalid text encoding. Export CSV as UTF-8 to preserve guest names.');
 let rows,legacy=false,isCSV=false;const warnings=[];text=text.replace(/^\uFEFF/,'');
 if(/\.json$/i.test(filename)||text.trim().startsWith('{')||text.trim().startsWith('[')){
  let p;try{p=JSON.parse(text);}catch{throw new Error('Invalid guest JSON.');}
  rows=Array.isArray(p)?p:p.guestList||p.guests;if(!Array.isArray(rows))throw new Error('No guest array found. Use a guest-table JSON or CSV file.');
  legacy=!p.format&&!p.guestList&&rows.some(g=>/^(L[1-5]|VIP[12]|ROUND[1-5]|EXTRA|HOLD|V[1-5])$/i.test(g?.tableId||''));
  if(p.format&&p.format!=='glass-house-guest-tables'&&p.format!=='tong-ter-extracted-guests')throw new Error('Unsupported guest file format.');
  if(p.format&&p.version!==1)throw new Error('Unsupported guest file version.');
  if(p.format==='tong-ter-extracted-guests')legacy=true;

 }else{
  isCSV=true;const csv=csvParse(text);if(!csv.length)throw new Error('CSV is empty.');const heads=csv.shift().map(h=>alias[key(h)]||'');if(!heads.includes('name'))throw new Error('CSV needs a name column.');const used=heads.filter(Boolean);if(new Set(used).size!==used.length)throw new Error('CSV contains duplicate column names.');
  rows=csv.map((r,i)=>{if(r.length!==heads.length)throw new Error('CSV row '+(i+2)+' has '+r.length+' fields; expected '+heads.length+'.');const o={};heads.forEach((h,j)=>{if(h)o[h]=r[j];});if(o.csvEscaped==='apostrophe-v1')for(const k of Object.keys(o))if(/^'[=+\-@\t\r']/.test(o[k]))o[k]=o[k].slice(1);return o;});
 }
 if(rows.length>5000)throw new Error('The guest list supports up to 5,000 guests.');if(!rows.length)throw new Error('The file contains no guests.');
 const ids=new Set(),normalized=[];let unknown=0;
 for(let i=0;i<rows.length;i++){
  const r=rows[i];if(!r||typeof r!=='object'||Array.isArray(r))throw new Error('Invalid guest at row '+(i+1));
  const g={id:String(r.id||uid())};for(const f of Object.keys(FIELDS))if(owns(r,f))g[f]=r[f];
  if(typeof g.name!=='string'||!g.name.trim())throw new Error('Guest name is required at row '+(i+1)+'.');
  if(ids.has(g.id))throw new Error('Duplicate guest ID in import: '+g.id);ids.add(g.id);
  for(const k of ['important','vip','reserve'])if(owns(r,k))g[k]=bool(r[k]);
  if(owns(r,'sourceSeatNumber'))g.sourceSeatNumber=numSeat(r.sourceSeatNumber,'Original seat');if(owns(r,'unmappedSeatNumber'))g.unmappedSeatNumber=numSeat(r.unmappedSeatNumber,'Unmatched imported seat');
  const hasTable=owns(r,'tableId')||owns(r,'tableLabel')||owns(r,'table_id')||owns(r,'table');
  const rawRef=isCSV&&owns(r,'tableLabel')?r.tableLabel:(r.tableId??r.table_id??r.tableLabel??r.table??''),rawSeat=r.seatNumber??r.seat;
  if(hasTable){let resolved=resolveTable(rawRef,s);if(resolved===null&&r.tableLabel&&/^table-|^$/.test(String(rawRef)))resolved=resolveTable(r.tableLabel,s);
   if(resolved===null){g.tableId='';g.seatNumber=null;g.unmappedTableId=String(rawRef);g.unmappedTableLabel=String(r.tableLabel||rawRef);g.unmappedSeatNumber=numSeat(rawSeat,'Unmatched imported seat');if(!g.sourceTableId)g.sourceTableId=String(rawRef);if(!g.sourceTableLabel)g.sourceTableLabel=String(r.tableLabel||rawRef);if(g.sourceSeatNumber==null)g.sourceSeatNumber=numSeat(rawSeat,'Original seat');unknown++;}
   else{g.tableId=resolved;g.seatNumber=resolved?numSeat(rawSeat):null;}
  }else if(owns(r,'seatNumber')||owns(r,'seat'))g.seatNumber=numSeat(rawSeat);
  // Validate partial text/flags now, without losing missing-column semantics for merge.
  clean({...g,tableId:'',seatNumber:null});normalized.push(g);
 }
 if(unknown)warnings.push(unknown+' guest(s) reference a table that does not exist here. Those references are saved as original tables; venue assignments will be unassigned.');
 return {rows:normalized,warnings,filename:String(filename),legacy};
}
